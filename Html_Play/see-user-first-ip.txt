<script>
document.addEventListener("DOMContentLoaded", function () {

    const checkIPButton = document.getElementById("check-ip-button");
    const loadingModal = document.getElementById("check-ip-loading");

    checkIPButton.addEventListener("click", async function () {

        const ip = prompt("Nhập IP cần kiểm tra:");

        // Người dùng bấm Cancel hoặc không nhập gì
        if (ip === null || ip.trim() === "") {
            return;
        }

        const inputIP = ip.trim();

        // Hiện modal loading
        loadingModal.style.display = "flex";

        try {

            const response = await fetch(
                "/Cover/CheckIPAtList?ip=" + encodeURIComponent(inputIP),
                {
                    method: "GET"
                }
            );

            if (!response.ok) {
                throw new Error("HTTP Status " + response.status);
            }

            const data = await response.json();

            let message = data.message || "";

            if (data.url) {
                message += "\nURL: " + data.url;
            }

            alert(message);

        } catch (error) {

            alert(
                "Không thể kiểm tra IP.\n" +
                "Lỗi: " + error.message
            );

        } finally {

            // Tắt modal loading dù API thành công hay lỗi
            loadingModal.style.display = "none";
        }
    });

});
</script>
