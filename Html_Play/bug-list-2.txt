function SeeTitle(data)
{
    for (var i = 0; i < data.length; i++)
    {
        var item = data[i];

        var row = document.getElementById("tr_" + item.Index);

        if (row == null)
        {
            continue;
        }

        row.style.cursor = "pointer";

        var nguyennhan = item.NguyenNhan;
        var giaiphap = item.GiaiPhap;

        var hasNguyenNhan =
            nguyennhan != null &&
            nguyennhan.toString().trim() !== "";

        var hasGiaiPhap =
            giaiphap != null &&
            giaiphap.toString().trim() !== "";

        if (!hasNguyenNhan && !hasGiaiPhap)
        {
            continue;
        }

        row.ondblclick = function ()
        {
            var oldModal = document.getElementById("myModalGlobal");

            if (oldModal)
            {
                oldModal.remove();
            }

            var html = "";

            if (hasNguyenNhan)
            {
                html += `
                    <div class="my-modal-section">
                        <h2 class="my-modal-title">
                            NGUYÊN NHÂN PHÁT SINH:
                        </h2>

                        <div class="my-modal-text">
                            ${nguyennhan}
                        </div>
                    </div>
                `;
            }

            if (hasGiaiPhap)
            {
                html += `
                    <div class="my-modal-section">
                        <h2 class="my-modal-title">
                            GIẢI PHÁP ĐÃ ÁP DỤNG:
                        </h2>

                        <div class="my-modal-text">
                            ${giaiphap}
                        </div>
                    </div>
                `;
            }

            var modal = document.createElement("div");

            modal.id = "myModalGlobal";

            modal.innerHTML = `
                <div class="my-modal-overlay">
                    <div class="my-modal-box">

                        <button
                            type="button"
                            class="my-modal-close">
                            ✕
                        </button>

                        <div class="my-modal-content">
                            ${html}
                        </div>

                    </div>
                </div>
            `;

            document.body.appendChild(modal);

            modal
                .querySelector(".my-modal-close")
                .onclick = function ()
                {
                    modal.remove();
                };
        };
    }
}
