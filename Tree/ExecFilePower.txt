        function getCurrentPathList() {
        
            const origin = window.location.origin.toLowerCase();
        
            if (origin.includes("mywebplayapi")) {
                return pathAPI;
            }
        
            if (origin.includes("normalwebbasic")) {
                return pathNormal;
            }
        
            if (origin.includes("submywebplay")) {

            return [];

        // ============================================================
        // ĐƯA OPTION VÀO SELECT
        // ============================================================
        
        function loadPathOptions() {
        
            const pathList = getCurrentPathList();
        
            pathSelect.innerHTML = "";
        
            if (pathList.length === 0) {
        
                const option = document.createElement("option");
        
                option.value = "";
                option.textContent = "Không có path";
        
                option.disabled = true;
                option.selected = true;
        
                pathSelect.appendChild(option);
        
                return;
            }
        
        
            pathList.forEach(function (item) {
        
                const option = document.createElement("option");
        
                option.value = item.value;
                option.textContent = item.key;
        
                pathSelect.appendChild(option);
        
            });
        }
